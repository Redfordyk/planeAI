## 💻 Опорный код фикстур

```python
@pytest.fixture
def make_chunk(db):
    import uuid, hashlib
    from ai.models import DocumentChunk
    def _make(workspace, content, project=None):
        seed = int(hashlib.md5(content.encode()).hexdigest(), 16)
        vec = [((seed >> i) & 0xFF)/255.0 for i in range(1536)]   # 1536!
        return DocumentChunk.objects.create(
            workspace=workspace, project=project, source_type="work_item",
            source_id=uuid.uuid4(), chunk_index=0, content=content,
            token_count=len(content.split()), embedding=vec,
            content_hash=hashlib.sha256(content.encode()).hexdigest())
    return _make
```
