## ✅ Что сделать

1. Создать приложение `ai`, зарегистрировать в `INSTALLED_APPS` (способ — из IMAGE.md, задача 0.5).
2. Реализовать модели:

```python
# ai/models.py
from django.db import models
from uuid import uuid4
from encrypted_model_fields.fields import EncryptedCharField
from pgvector.django import VectorField, HnswIndex

class DocumentChunk(models.Model):
    id           = models.UUIDField(primary_key=True, default=uuid4)
    workspace    = models.ForeignKey("<app>.Workspace", on_delete=models.CASCADE)  # из SCHEMA.md
    project      = models.ForeignKey("<app>.Project", on_delete=models.CASCADE, null=True)
    source_type  = models.CharField(max_length=20)   # work_item | comment | page
    source_id    = models.UUIDField()
    chunk_index  = models.IntegerField()
    content      = models.TextField()
    token_count  = models.IntegerField()
    embedding    = VectorField(dimensions=1536)       # text-embedding-3-small
    content_hash = models.CharField(max_length=64)
    updated_at   = models.DateTimeField(auto_now=True)
    class Meta:
        indexes = [
            HnswIndex(name="chunk_hnsw_idx", fields=["embedding"],
                      m=16, ef_construction=64, opclasses=["vector_cosine_ops"]),
            models.Index(fields=["workspace", "project"]),
            models.Index(fields=["source_type", "source_id"]),
        ]

class WorkspaceAIConfig(models.Model):
    workspace            = models.OneToOneField("<app>.Workspace", on_delete=models.CASCADE)
    anthropic_key        = EncryptedCharField(max_length=255)
    openai_key           = EncryptedCharField(max_length=255)
    chat_model           = models.CharField(max_length=60, default="claude-sonnet-4-6")
    embed_model          = models.CharField(max_length=64, default="text-embedding-3-small")
    monthly_token_budget = models.BigIntegerField(default=5_000_000)
    enabled              = models.BooleanField(default=False)

class AIUsageLog(models.Model):
    id           = models.UUIDField(primary_key=True, default=uuid4)
    workspace    = models.ForeignKey("<app>.Workspace", on_delete=models.CASCADE)
    user         = models.ForeignKey("<app>.User", on_delete=models.SET_NULL, null=True)
    feature      = models.CharField(max_length=40)   # intent_search|summarize|bulk|agent|embed
    model        = models.CharField(max_length=60)
    input_tokens          = models.IntegerField(default=0)
    output_tokens         = models.IntegerField(default=0)
    cache_read_tokens     = models.IntegerField(default=0)
    cache_creation_tokens = models.IntegerField(default=0)
    cost_usd     = models.DecimalField(max_digits=10, decimal_places=6, default=0)
    created_at   = models.DateTimeField(auto_now_add=True, db_index=True)
    class Meta:
        indexes = [models.Index(fields=["workspace", "created_at"])]

class AIProjectSettings(models.Model):
    project         = models.OneToOneField("<app>.Project", on_delete=models.CASCADE)
    exclude_from_ai = models.BooleanField(default=False)
```

3. Зарегистрировать модели в Django admin (для удобной настройки ключей).
